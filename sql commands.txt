
CREATE DATABASE admin_system;
USE admin_system;
CREATE TABLE admins (
    id INT UNSIGNED NOT NULL AUTO_INCREMENT,
    username VARCHAR(50) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    pin VARCHAR(6) NOT NULL,
    PRIMARY KEY (id)
);

CREATE TABLE hospital_system.doctors_schedule (
    id INT AUTO_INCREMENT PRIMARY KEY,
    doctor_id VARCHAR(10) NOT NULL, -- Match the data type of doctorID
    day VARCHAR(10) NOT NULL,
    time_slot TIME NOT NULL,
    status ENUM('available', 'booked') DEFAULT 'available',
    FOREIGN KEY (doctor_id) REFERENCES doctors(doctorID) ON DELETE CASCADE ON UPDATE CASCADE
);
